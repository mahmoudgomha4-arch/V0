let index = 0, answers = Array(20).fill(null), time = 40 * 60, timerInterval;

function toggleTheme() { document.body.classList.toggle('dark-mode'); }

function showAdminLogin() { 
    document.getElementById('admin-login-screen').classList.remove('hidden'); 
}

function closeScreens() { 
    document.getElementById('admin-login-screen').classList.add('hidden');
    document.getElementById('admin-panel').classList.add('hidden');
}

function loginAdmin() {
    if(document.getElementById('adminCode').value === "nannannan") {
        document.getElementById('admin-login-screen').classList.add('hidden');
        document.getElementById('admin-panel').classList.remove('hidden');
        loadTable();
    } else { alert("كود خاطئ!"); }
}

function loadTable() {
    const res = JSON.parse(localStorage.getItem('tofas_db') || "[]");
    document.querySelector("#resultsTable tbody").innerHTML = res.map(r => `<tr><td>${r.name}</td><td>${r.phone}</td><td>${r.score}</td></tr>`).join('');
}

function resetMyAdmin() {
    const name = localStorage.getItem('last_name');
    if(!name) return alert("لا توجد نتيجة مسجلة حالياً!");
    let db = JSON.parse(localStorage.getItem('tofas_db') || "[]");
    localStorage.setItem('tofas_db', JSON.stringify(db.filter(r => r.name !== name)));
    localStorage.removeItem('finished');
    alert("تم حذف نتيجتك.. الصفحة ستفتح لك الآن.");
    location.reload();
}

function startQuiz() {
    const name = document.getElementById('username').value.trim();
    const phone = document.getElementById('userphone').value.trim();
    if(!name || !phone) return alert("الرجاء إدخال الاسم ورقم الهاتف!");
    
    localStorage.setItem('temp_name', name);
    localStorage.setItem('temp_phone', phone);
    document.getElementById('welcome-screen').classList.add('hidden');
    render();
    
    timerInterval = setInterval(() => {
        let m = Math.floor(time/60), s = time%60;
        document.getElementById('timer').innerText = `${m}:${s < 10 ? '0' : ''}${s}`;
        if (time <= 0) finish();
        time--;
    }, 1000);
}

function render() {
    const d = quizData[index];
    document.getElementById('qText').innerText = d.q;
    document.getElementById('qCode').innerText = d.code;
    document.getElementById('qOptions').innerHTML = d.options.map((o, i) => `
        <label class="opt-label">
            <input type="radio" name="ans" ${answers[index] === i ? 'checked' : ''} onclick="answers[${index}]=${i}">
            <span style="margin-right:10px">${o}</span>
        </label>
    `).join('');
    document.getElementById('nextBtn').innerText = index === 19 ? "إنهاء" : "التالي";
}

function next() { if(index < 19) { index++; render(); } else finish(); }
function prev() { if(index > 0) { index--; render(); } }

function finish() {
    clearInterval(timerInterval);
    let s = 0;
    quizData.forEach((q, i) => { if(answers[i] === q.answer) s++; });
    const scoreStr = (s/20)*100 + "%";
    
    let db = JSON.parse(localStorage.getItem('tofas_db') || "[]");
    db.push({ name: localStorage.getItem('temp_name'), phone: localStorage.getItem('temp_phone'), score: scoreStr });
    localStorage.setItem('tofas_db', JSON.stringify(db));
    localStorage.setItem('finished', 'true');
    localStorage.setItem('last_score', scoreStr);
    localStorage.setItem('last_name', localStorage.getItem('temp_name'));

    showResults(localStorage.getItem('temp_name'), scoreStr);
}

function showResults(n, s) {
    document.getElementById('welcome-screen').classList.add('hidden');
    document.getElementById('result-screen').classList.remove('hidden');
    document.getElementById('final-name').innerText = "المتسابق: " + n;
    document.getElementById('final-score').innerText = "الدرجة: " + s;
    if(s === "100%") document.getElementById('gift-area').innerText = "🎁 ممتاز! تواصل مع محمود: 01283425292";
}

window.onload = () => {
    if(localStorage.getItem('finished')) {
        showResults(localStorage.getItem('last_name'), localStorage.getItem('last_score'));
    }
};
